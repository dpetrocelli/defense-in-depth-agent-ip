\node[layer, fill=l2fill, draw=l2line, right=3mm of l1] (l2) {\textbf{Layer 2}\\\textbf{Input validation}\\[2pt] risk score, four severity tiers\\ typoglycemia detector\\ encoding detector\\ sanitization; score $\geq 20$: block};

\node[draw, rounded corners=2pt, align=center, inner sep=3pt, fill=hardfill, draw=hardline, above=3.5mm of llm, text width=3.2cm] (hard) {\textbf{Prompt hardening}\\defensive suffix + canary tokens};
